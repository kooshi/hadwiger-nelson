def HadwigerNelson.Rigidity.NoRamifiedStableDyadic : Subfield ℝ → Prop :=
fun F =>
  ∀ (P : HadwigerNelson.SpectralChain.StablePrime F),
    HadwigerNelson.Rigidity.IsDyadic P → ¬HadwigerNelson.Rigidity.IsRamified P
def HadwigerNelson.Rigidity.IsRamified : {F : Subfield ℝ} → HadwigerNelson.SpectralChain.StablePrime F → Prop :=
fun {F} P =>
  ∀ (x : NumberField.RingOfIntegers ↥(HadwigerNelson.SpectralChain.fieldI F)),
    (HadwigerNelson.SpectralChain.conjO F) x - x ∈ P.P
def HadwigerNelson.Rigidity.IsDyadic : {F : Subfield ℝ} → HadwigerNelson.SpectralChain.StablePrime F → Prop :=
fun {F} P => 2 ∈ P.P
def HadwigerNelson.Rigidity.stableIntegers : (F : Subfield ℝ) →
  [NumberField ↥F] → Subring ↥(HadwigerNelson.SpectralChain.fieldI F) :=
fun F [NumberField ↥F] => ⨅ P, P.localRing.toSubring
def HadwigerNelson.UnitRing.unitRing.{u_1} : {L : Type u_1} → [inst : Field L] → (L →+* L) → Subring L :=
fun {L} [Field L] σ => Subring.closure (HadwigerNelson.normOne σ)
def HadwigerNelson.UnitRing.unitRingOver : Subfield ℝ → Subring ℂ :=
fun F => Subring.closure {z | z ∈ HadwigerNelson.planeOver F ∧ ‖z‖ = 1}
structure HadwigerNelson.SpectralChain.StablePrime (F : Subfield ℝ) : Type
number of parameters: 1
fields:
  HadwigerNelson.SpectralChain.StablePrime.P : Ideal
      (NumberField.RingOfIntegers ↥(HadwigerNelson.SpectralChain.fieldI F))

  HadwigerNelson.SpectralChain.StablePrime.comap_conj : Ideal.comap (HadwigerNelson.SpectralChain.conjO F) self.P =
      self.P
constructor:
  HadwigerNelson.SpectralChain.StablePrime.mk {F : Subfield ℝ}
    (P : Ideal (NumberField.RingOfIntegers ↥(HadwigerNelson.SpectralChain.fieldI F))) (isMaximal : P.IsMaximal)
    (comap_conj : Ideal.comap (HadwigerNelson.SpectralChain.conjO F) P = P) : HadwigerNelson.SpectralChain.StablePrime F
def HadwigerNelson.SpectralChain.localBound : Subfield ℝ → ℕ∞ :=
fun F => ⨅ P, (HadwigerNelson.SpectralChain.levelTarget P 1).chromaticNumber
def HadwigerNelson.Arith.IsUsablePrime : {F : Subfield ℝ} → Ideal (NumberField.RingOfIntegers ↥F) → Prop :=
fun {F} 𝔭 => 𝔭.IsMaximal ∧ Nat.card (NumberField.RingOfIntegers ↥F ⧸ 𝔭) % 4 = 3
