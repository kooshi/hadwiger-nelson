def HadwigerNelson.Plane.IsCommonUnitPoint : ℂ → ℂ → ℂ → Prop :=
fun x y w => dist w x = 1 ∧ dist w y = 1
def HadwigerNelson.Plane.ThreePiece.SeparatesAt.{u_1} : {α : Type u_1} →
  {U : Set ℂ} → {c : ℂ → α} → {x : ℂ} → ThreePiece U c x → ℂ → Prop :=
fun {α} {U} {c} {x} D w => ∃ i j, InnerAt (D.piece i) w ∧ OuterAt (D.piece j) w
def HadwigerNelson.Plane.ThreePiece.OneOfEach.{u_1} : {α : Type u_1} →
  {U : Set ℂ} → {c : ℂ → α} → {x : ℂ} → ThreePiece U c x → ℂ → Equiv.Perm (Fin 3) → Prop :=
fun {α} {U} {c} {x} D w σ => InnerAt (D.piece (σ 0)) w ∧ OuterAt (D.piece (σ 1)) w ∧ CrossingAt (D.piece (σ 2)) w
def HadwigerNelson.Plane.ThreePiece.IsFull.{u_1} : {α : Type u_1} →
  {U : Set ℂ} → {c : ℂ → α} → {x : ℂ} → ThreePiece U c x → Prop :=
fun {α} {U} {c} {x} D => ∀ (i j : Fin 3), i ≠ j → x ∈ closure (D.edge i j \ {x})
def HadwigerNelson.Plane.ThreePiece.HasArcEdges.{u_1} : {α : Type u_1} →
  {U : Set ℂ} → {c : ℂ → α} → {x : ℂ} → ThreePiece U c x → Prop :=
fun {α} {U} {c} {x} D => ∀ (i j : Fin 3), i ≠ j → ∃ γ, Function.Injective ⇑γ ∧ γ 0 = x ∧ Set.range ⇑γ ⊆ D.edge i j
def HadwigerNelson.Plane.ThreePiece.Saves.{u_1} : {α : Type u_1} →
  {U : Set ℂ} → {c : ℂ → α} → {x : ℂ} → ThreePiece U c x → ℂ → Prop :=
fun {α} {U} {c} {x} D w => ∃ i j, i ≠ j ∧ HasUnitArcFrom (D.edge i j) w x
def HadwigerNelson.Plane.HasUnitArcFrom : Set ℂ → ℂ → ℂ → Prop :=
fun E w x => ∃ s, (s = 1 ∨ s = -1) ∧ ∃ δ > 0, ∀ t ∈ Set.Icc 0 δ, w + (x - w) * e (s * t) ∈ E
def HadwigerNelson.Plane.ThreePiece.edge.{u_1} : {α : Type u_1} →
  {U : Set ℂ} → {c : ℂ → α} → {x : ℂ} → ThreePiece U c x → Fin 3 → Fin 3 → Set ℂ :=
fun {α} {U} {c} {x} D i j => closure (D.piece i) ∩ closure (D.piece j) ∩ D.nbhd
def HadwigerNelson.Plane.presentAt.{u_1} : {α : Type u_1} → Set ℂ → (ℂ → α) → ℂ → Set α :=
fun {α} U c w => {i | w ∈ essSupport (U ∩ c ⁻¹' {i})}
def HadwigerNelson.Plane.InnerAt : Set ℂ → ℂ → Prop :=
fun P w => P ⊆ Metric.closedBall w 1
def HadwigerNelson.Plane.OuterAt : Set ℂ → ℂ → Prop :=
fun P w => Disjoint P (Metric.ball w 1)
def HadwigerNelson.Plane.CrossingAt : Set ℂ → ℂ → Prop :=
fun P w => ¬InnerAt P w ∧ ¬OuterAt P w
