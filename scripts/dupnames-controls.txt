# The doubled-name gate's controls (FS131-T4, 2026-10-06) — read by `./scripts/dupnames.sh --self-test`.
#
# A doubled name needs two compiled modules, so the plants are a library, not lines: the `## module` blocks below are written to
# the scratch Lake package `data/fs131/t4/plant/` (core Lean only; two roots, `T4Plant` — the package under audit — and `T4Dep`,
# standing for a dependency) and built there. Each `## control` then elaborates a COPY of the shipped `scripts/DupNames.lean` whose
# `import Proofs` is replaced by the control's imports and whose root is the control's, and the gate's own `check_scan` judges what
# it printed. Every check the gate makes has a control that must make it fire, and the PASS controls are the partners showing a
# refusal comes from the plant and not from the scaffolding.
#
# Directives:
#   ## module <Module>          the module's text follows, up to the next `## ` line
#   ## control <name>           a control; then
#   ## import <Module> …        the scan copy's imports, in order (the import order is the order the scan reports copies in)
#   ## root <Name>              the package root the scan is told (default `T4Plant`)
#   ## sources <n>              the number of source files the package is said to have — what the coverage check compares with
#   ## drop-output <prefix>     delete the first output line with that prefix, after elaboration (a lost line, a run cut short)
#   ## expect PASS | FAIL <kind> <text> | NOFAIL <kind> | COUNT <key>=<value> | ELAB <text>
#                               PASS: elaborates and no FAIL line; FAIL: a `FAIL <kind>` line containing <text>; NOFAIL: no such
#                               line; COUNT: the COUNTS line holds it; ELAB: the copy does not elaborate, and prints <text>
# A control is two-sided: every FAIL kind it raises must be one it predicts. Lines starting `# ` outside a block are comments.
# No module here holds a `sorry`: a copy that is not clean rests on a planted `axiom` instead (the `sorry` rows are the
# experiment's, `data/fs131/t4/experiment-table.md`).

## module T4Plant.Base
namespace T4P
/-- A two-element type, for the instance pair. -/
inductive Tok where
  | a
  | b
/-- A recursive function, so that Lean generates its equation lemma on demand. -/
def fn : Nat → Nat
  | 0 => 1
  | n + 1 => fn n
end T4P

## module T4Plant.Clean
import T4Plant.Base
theorem T4P.only_here : 2 + 2 = 4 := rfl

## module T4Plant.DupA
import T4Plant.Base
theorem T4P.dup : 2 + 2 = 4 := rfl

## module T4Plant.DupB
import T4Plant.Base
theorem T4P.dup : 2 + 2 = 4 := Eq.refl 4

## module T4Plant.AxA
import T4Plant.Base
axiom T4P.planted_ax : 3 + 3 = 6
theorem T4P.dup_ax : 3 + 3 = 6 := T4P.planted_ax

## module T4Plant.AxB
import T4Plant.Base
theorem T4P.dup_ax : 3 + 3 = 6 := rfl

## module T4Plant.KindA
import T4Plant.Base
axiom T4P.dup_kind : 4 + 4 = 8

## module T4Plant.KindB
import T4Plant.Base
theorem T4P.dup_kind : 4 + 4 = 8 := rfl

## module T4Plant.InstA
import T4Plant.Base
namespace T4P
instance : Nonempty Tok := ⟨.a⟩
end T4P

## module T4Plant.InstB
import T4Plant.Base
namespace T4P
instance : Nonempty Tok := ⟨.b⟩
end T4P

## module T4Plant.GenA
import T4Plant.Base

## module T4Plant.GenB
import T4Plant.Base
theorem T4P.ask_b : T4P.fn 0 = 1 := T4P.fn.eq_1

## module T4Plant.Shared
theorem T4P.shared : 5 + 5 = 10 := Eq.refl 10

## module T4Dep.X
theorem T4P.shared : 5 + 5 = 10 := rfl
theorem T4P.theirs : 6 + 6 = 12 := rfl

## module T4Dep.Z
theorem T4P.theirs : 6 + 6 = 12 := Eq.refl 12

# ---- the class that is not a failure, and its limits -------------------------------------------------------------------------

# A theorem Lean generates on demand, asked for in two modules: counted, listed, and no failure. Asking for `fn.eq_1` makes Lean
# generate the family — `T4P.fn.eq_1`, `T4P.fn.eq_2`, `T4P.fn.eq_def` — in each asking module, so three names repeat (measured at
# this control's first run, which expected one; and `eq_def`'s two copies are not one term, so the class cannot ask for that).
## control generated_only
## import T4Plant.GenA T4Plant.GenB
## sources 3
## expect PASS
## expect COUNT generated=3
## expect COUNT doubled=0

# The exclusion does not shelter a hand-written pair standing beside a generated one.
## control generated_beside_doubled
## import T4Plant.GenA T4Plant.GenB T4Plant.DupA T4Plant.DupB
## sources 5
## expect FAIL doubled T4P.dup
## expect COUNT generated=3
## expect COUNT doubled=1
